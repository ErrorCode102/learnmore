import { certificateLessons, diplomaLessons, getLessonsForLevel } from '../data/course';
import type { LessonId, Level, ProgressState } from '../types/course';

const STORAGE_KEY = 'auth-classroom:progress:v1';
const LEVELS: Level[] = ['certificate', 'diploma'];

function createDefaultProgress(): ProgressState {
  return {
    version: 1,
    level: 'certificate',
    completed: {
      certificate: [],
      diploma: [],
    },
    quizScores: {},
  };
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function isLevel(value: unknown): value is Level {
  return value === 'certificate' || value === 'diploma';
}

function normalizeCompleted(level: Level, value: unknown): LessonId[] {
  if (!Array.isArray(value)) {
    return [];
  }

  const allowedIds = new Set(getLessonsForLevel(level).map((lesson) => lesson.id));
  const seen = new Set<LessonId>();
  const completed: LessonId[] = [];

  for (const id of value) {
    if (typeof id !== 'string') {
      continue;
    }

    const lessonId = id as LessonId;
    if (!allowedIds.has(lessonId) || seen.has(lessonId)) {
      continue;
    }

    seen.add(lessonId);
    completed.push(lessonId);
  }

  return completed;
}

function normalizeQuizScores(value: unknown): Partial<Record<Level, number>> {
  if (!isRecord(value)) {
    return {};
  }

  const quizScores: Partial<Record<Level, number>> = {};

  for (const level of LEVELS) {
    const score = value[level];
    if (
      typeof score === 'number' &&
      Number.isInteger(score) &&
      score >= 0 &&
      score <= 5
    ) {
      quizScores[level] = score;
    }
  }

  return quizScores;
}

function normalizeProgress(value: unknown): ProgressState {
  if (!isRecord(value) || value.version !== 1 || !isLevel(value.level)) {
    return createDefaultProgress();
  }

  const completedInput = isRecord(value.completed) ? value.completed : {};
  const completed = {
    certificate: normalizeCompleted('certificate', completedInput.certificate),
    diploma: normalizeCompleted('diploma', completedInput.diploma),
  };
  const progress: ProgressState = {
    version: 1,
    level: value.level,
    completed,
    quizScores: normalizeQuizScores(value.quizScores),
  };
  const allowedLastLessonIds = new Set(
    [...certificateLessons, ...diplomaLessons].map((lesson) => lesson.id),
  );

  if (
    typeof value.lastLessonId === 'string' &&
    allowedLastLessonIds.has(value.lastLessonId as LessonId)
  ) {
    progress.lastLessonId = value.lastLessonId as LessonId;
  }

  return progress;
}

export function readProgress(): ProgressState {
  if (typeof window === 'undefined') {
    return createDefaultProgress();
  }

  try {
    const stored = window.localStorage.getItem(STORAGE_KEY);
    return stored ? normalizeProgress(JSON.parse(stored)) : createDefaultProgress();
  } catch {
    return createDefaultProgress();
  }
}

export function saveProgress(next: ProgressState): void {
  if (typeof window === 'undefined') {
    return;
  }

  try {
    window.localStorage.setItem(STORAGE_KEY, JSON.stringify(normalizeProgress(next)));
  } catch {
    // Storage can be unavailable in private browsing or when its quota is exceeded.
  }
}

export function resetProgress(): ProgressState {
  const progress = createDefaultProgress();
  saveProgress(progress);
  return progress;
}

export function toggleLesson(
  state: ProgressState,
  id: LessonId,
  level: Level,
): ProgressState {
  const lessons = getLessonsForLevel(level);
  if (!lessons.some((lesson) => lesson.id === id)) {
    throw new Error(`Unknown lesson "${id}" for ${level} level.`);
  }

  const completedForLevel = state.completed[level];
  const isCompleted = completedForLevel.includes(id);
  const completed = isCompleted
    ? completedForLevel.filter((lessonId) => lessonId !== id)
    : [...completedForLevel, id];

  return {
    ...state,
    level,
    completed: {
      ...state.completed,
      [level]: completed,
    },
    lastLessonId: id,
  };
}

export function recordQuiz(
  state: ProgressState,
  level: Level,
  score: number,
): ProgressState {
  const integerScore = Number.isNaN(score) ? 0 : Math.trunc(score);
  const clampedScore = Math.max(0, Math.min(5, integerScore));

  return {
    ...state,
    level,
    quizScores: {
      ...state.quizScores,
      [level]: clampedScore,
    },
  };
}

import { fireEvent, render, screen } from '@testing-library/react';
import { beforeEach, describe, expect, it, vi } from 'vitest';
import { useProgress } from '../hooks/useProgress';
import { readProgress, saveProgress } from '../lib/progress';
import type { ProgressState } from '../types/course';
import { LevelSwitcher } from './LevelSwitcher';

beforeEach(() => {
  localStorage.clear();
});

describe('LevelSwitcher', () => {
  it('renders the exact track labels with accessible selected state', () => {
    const onChange = vi.fn();
    const { rerender } = render(
      <LevelSwitcher level="certificate" onChange={onChange} />,
    );

    const certificate = screen.getByRole('button', { name: 'ปวช.' });
    const diploma = screen.getByRole('button', { name: 'ปวส.' });

    expect(certificate).toHaveAttribute('aria-pressed', 'true');
    expect(certificate).toHaveClass('bg-ink', 'text-surface');
    expect(diploma).toHaveAttribute('aria-pressed', 'false');

    fireEvent.click(diploma);

    expect(onChange).toHaveBeenCalledWith('diploma');

    rerender(<LevelSwitcher level="diploma" onChange={onChange} />);

    expect(certificate).toHaveAttribute('aria-pressed', 'false');
    expect(diploma).toHaveAttribute('aria-pressed', 'true');
    expect(diploma).toHaveClass('bg-ink', 'text-surface');
  });

  it('preserves a diploma-only last lesson when switching to certificate', () => {
    const initial: ProgressState = {
      version: 1,
      level: 'diploma',
      completed: {
        certificate: ['overview'],
        diploma: ['security'],
      },
      quizScores: {
        certificate: 4,
        diploma: 5,
      },
      lastLessonId: 'security',
    };
    saveProgress(initial);

    function ProgressHarness() {
      const { level, setLevel } = useProgress();

      return <LevelSwitcher level={level} onChange={setLevel} />;
    }

    render(<ProgressHarness />);
    fireEvent.click(screen.getByRole('button', { name: 'ปวช.' }));

    expect(screen.getByRole('button', { name: 'ปวช.' })).toHaveAttribute(
      'aria-pressed',
      'true',
    );
    expect(readProgress()).toEqual({
      ...initial,
      level: 'certificate',
    });
  });

  it('switches the persisted level without clearing either track progress', () => {
    const initial: ProgressState = {
      version: 1,
      level: 'certificate',
      completed: {
        certificate: ['overview', 'api'],
        diploma: ['security'],
      },
      quizScores: {
        certificate: 4,
        diploma: 5,
      },
      lastLessonId: 'api',
    };
    saveProgress(initial);

    function ProgressHarness() {
      const { level, setLevel } = useProgress();

      return <LevelSwitcher level={level} onChange={setLevel} />;
    }

    render(<ProgressHarness />);
    fireEvent.click(screen.getByRole('button', { name: 'ปวส.' }));

    expect(screen.getByRole('button', { name: 'ปวส.' })).toHaveAttribute(
      'aria-pressed',
      'true',
    );
    expect(readProgress()).toEqual({
      ...initial,
      level: 'diploma',
    });
  });
});
