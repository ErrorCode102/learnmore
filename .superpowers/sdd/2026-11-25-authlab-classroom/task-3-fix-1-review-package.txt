import { ArrowRight, BookOpenCheck, Clock3 } from 'lucide-react';
import { Link } from 'react-router-dom';
import { useClassroomProgress } from '../context/ProgressContext';
import type { LessonId, LessonSummary } from '../types/course';

export interface ContinueSelection {
  lesson: LessonSummary;
  label: 'เริ่มบทแรก' | 'เรียนต่อ' | 'ทบทวนบทสุดท้าย';
}

export function getContinueSelection(
  lessons: LessonSummary[],
  completed: LessonId[],
): ContinueSelection | undefined {
  const finalLesson = lessons.at(-1);
  if (!finalLesson) {
    return undefined;
  }

  const completedIds = new Set(completed);
  const allComplete = lessons.every((lesson) => completedIds.has(lesson.id));
  const lesson =
    lessons.find((candidate) => !completedIds.has(candidate.id)) ?? finalLesson;

  if (allComplete) {
    return { lesson: finalLesson, label: 'ทบทวนบทสุดท้าย' };
  }

  return {
    lesson,
    label: completed.length === 0 ? 'เริ่มบทแรก' : 'เรียนต่อ',
  };
}

export interface ContinueCardProps {
  lessons: LessonSummary[];
}

export function ContinueCard({ lessons }: ContinueCardProps) {
  const { level, progress } = useClassroomProgress();
  const completed = progress.completed[level];
  const selection = getContinueSelection(lessons, completed);

  if (!selection) {
    return null;
  }

  return (
    <section
      aria-labelledby="continue-title"
      className="relative overflow-hidden rounded-[1.75rem] bg-ink p-6 text-white shadow-[0_24px_60px_-32px_rgba(23,34,59,0.7)] sm:p-7"
    >
      <div
        aria-hidden="true"
        className="absolute -right-14 -top-16 h-44 w-44 rounded-full bg-brand/35 blur-3xl"
      />
      <div className="relative">
        <div className="flex items-center gap-3">
          <span className="flex h-11 w-11 items-center justify-center rounded-2xl bg-white/10 text-coral">
            <BookOpenCheck aria-hidden="true" className="h-5 w-5" strokeWidth={1.8} />
          </span>
          <div>
            <p className="text-[10px] font-semibold tracking-[0.18em] text-white/55">
              YOUR NEXT STEP
            </p>
            <h2 id="continue-title" className="mt-1 text-lg font-semibold">
              เส้นทางถัดไปของคุณ
            </h2>
          </div>
        </div>

        <p className="mt-6 text-xl font-semibold leading-8 tracking-[-0.02em]">
          {selection.lesson.title}
        </p>
        <p className="mt-2 flex items-center gap-2 text-sm text-white/65">
          <Clock3 aria-hidden="true" className="h-4 w-4" />
          บทที่ {selection.lesson.order} · {selection.lesson.durationMinutes} นาที
        </p>

        <Link
          to={`/lesson/${selection.lesson.id}`}
          className="group mt-6 inline-flex min-h-12 w-full items-center justify-center gap-3 rounded-xl bg-coral px-5 py-3 text-sm font-semibold text-ink transition-colors hover:bg-[#e85f50] focus-visible:outline-white sm:w-auto"
        >
          {selection.label}
          <ArrowRight
            aria-hidden="true"
            className="h-4 w-4 transition-transform group-hover:translate-x-0.5"
          />
        </Link>
      </div>
    </section>
  );
}

interface Rgb {
  r: number;
  g: number;
  b: number;
}

import sourceStyles from '../index.css?raw';

function parseHex(color: string): Rgb {
  const hex = color.trim().replace('#', '');
  if (!/^[\da-f]{6}$/i.test(hex)) {
    throw new Error(`Expected a six-digit hex color, received ${color}.`);
  }
  return {
    r: Number.parseInt(hex.slice(0, 2), 16),
    g: Number.parseInt(hex.slice(2, 4), 16),
    b: Number.parseInt(hex.slice(4, 6), 16),
  };
}

function relativeLuminance(color: Rgb) {
  const channels = [color.r, color.g, color.b].map((channel) => {
    const value = channel / 255;
    return value <= 0.04045 ? value / 12.92 : ((value + 0.055) / 1.055) ** 2.4;
  });
  return channels[0] * 0.2126 + channels[1] * 0.7152 + channels[2] * 0.0722;
}

function contrastRatio(foreground: string, background: string) {
  const foregroundLuminance = relativeLuminance(parseHex(foreground));
  const backgroundLuminance = relativeLuminance(parseHex(background));
  const lighter = Math.max(foregroundLuminance, backgroundLuminance);
  const darker = Math.min(foregroundLuminance, backgroundLuminance);
  return (lighter + 0.05) / (darker + 0.05);
}

function themeColor(name: 'ink' | 'coral') {
  const css = sourceStyles;
  const value = css.match(new RegExp(`--${name}:\\s*(#[\\da-fA-F]{6})`))?.[1];
  if (!value) {
    throw new Error(`Theme color --${name} was not found.`);
  }
  return value;
}

function hoverBackground(cta: HTMLElement) {
  const hoverClass = Array.from(cta.classList).find((className) =>
    className.startsWith('hover:bg-[#'),
  );
  const color = hoverClass?.match(/#[\da-fA-F]{6}/)?.[0];
  if (!color) {
    throw new Error('Expected the CTA to declare a six-digit hex hover background.');
  }
  return color;
}

function ContinueLink() {
  return (
    <MemoryRouter>
      <ProgressProvider>
        <ContinueCard lessons={certificateLessons} />
      </ProgressProvider>
    </MemoryRouter>
  );
}

beforeEach(() => {
  localStorage.clear();
});

describe('ContinueCard CTA contrast', () => {
  it('uses dark ink on the coral background in normal and hover treatments', () => {
    expect(themeColor('coral')).toBe('#f97362');
    expect(contrastRatio(themeColor('ink'), themeColor('coral'))).toBeGreaterThanOrEqual(4.5);
    expect(contrastRatio(themeColor('ink'), '#e85f50')).toBeGreaterThanOrEqual(4.5);
  });

  it('keeps the rendered CTA text ink at AA contrast for its actual hover background', () => {
    render(<ContinueLink />);
    const cta = screen.getByRole('link', { name: 'เริ่มบทแรก' });
    const hoverColor = hoverBackground(cta);

    expect(cta).toHaveClass('bg-coral', 'text-ink');
    expect(cta).not.toHaveClass('text-white');
    expect(contrastRatio(themeColor('ink'), themeColor('coral'))).toBeGreaterThanOrEqual(4.5);
    expect(contrastRatio(themeColor('ink'), hoverColor)).toBeGreaterThanOrEqual(4.5);

    fireEvent.mouseOver(cta);
    expect(cta).toHaveClass('text-ink');
    expect(hoverBackground(cta)).toBe(hoverColor);
  });
});

    const scrollTo = vi.spyOn(window, 'scrollTo').mockImplementation(() => undefined);

    const scrollTo = vi.spyOn(window, 'scrollTo').mockImplementation(() => undefined);
    renderLesson('overview');

    try {
      scrollTo.mockClear();
      const breadcrumbs = screen.getByRole('navigation', { name: 'เส้นทางนำทาง' });
      const breadcrumb = within(breadcrumbs).getByRole('link', {
        name: 'กลับไปยังแผนภูมิหลักสูตร',

      expect(breadcrumb).toHaveAttribute('href', '#/#course-map');
      fireEvent.click(breadcrumb);

      await waitFor(() => {
        expect(document.getElementById('course-map')).toHaveFocus();

      });
      expect(window.location.hash).toBe('#/#course-map');
