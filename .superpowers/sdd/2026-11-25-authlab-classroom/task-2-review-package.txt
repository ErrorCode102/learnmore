import { getLessonsForLevel } from '../data/course';
import type { LessonId, Level, ProgressState } from '../types/course';

const STORAGE_KEY = 'auth-classroom:progress:v1';
const LEVELS: Level[] = ['certificate', 'diploma'];

function createDefaultProgress(): ProgressState {
  return {
    version: 1,
    level: 'certificate',
    completed: {
      certificate: [],
      diploma: [],
    },
    quizScores: {},
  };
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function isLevel(value: unknown): value is Level {
  return value === 'certificate' || value === 'diploma';
}

function normalizeCompleted(level: Level, value: unknown): LessonId[] {
  if (!Array.isArray(value)) {
    return [];
  }

  const allowedIds = new Set(getLessonsForLevel(level).map((lesson) => lesson.id));
  const seen = new Set<LessonId>();
  const completed: LessonId[] = [];

  for (const id of value) {
    if (typeof id !== 'string') {
      continue;
    }

    const lessonId = id as LessonId;
    if (!allowedIds.has(lessonId) || seen.has(lessonId)) {
      continue;
    }

    seen.add(lessonId);
    completed.push(lessonId);
  }

  return completed;
}

function normalizeQuizScores(value: unknown): Partial<Record<Level, number>> {
  if (!isRecord(value)) {
    return {};
  }

  const quizScores: Partial<Record<Level, number>> = {};

  for (const level of LEVELS) {
    const score = value[level];
    if (
      typeof score === 'number' &&
      Number.isInteger(score) &&
      score >= 0 &&
      score <= 5
    ) {
      quizScores[level] = score;
    }
  }

  return quizScores;
}

function normalizeProgress(value: unknown): ProgressState {
  if (!isRecord(value) || value.version !== 1 || !isLevel(value.level)) {
    return createDefaultProgress();
  }

  const completedInput = isRecord(value.completed) ? value.completed : {};
  const completed = {
    certificate: normalizeCompleted('certificate', completedInput.certificate),
    diploma: normalizeCompleted('diploma', completedInput.diploma),
  };
  const progress: ProgressState = {
    version: 1,
    level: value.level,
    completed,
    quizScores: normalizeQuizScores(value.quizScores),
  };
  const allowedLastLessonIds = new Set(
    getLessonsForLevel(value.level).map((lesson) => lesson.id),
  );

  if (
    typeof value.lastLessonId === 'string' &&
    allowedLastLessonIds.has(value.lastLessonId as LessonId)
  ) {
    progress.lastLessonId = value.lastLessonId as LessonId;
  }

  return progress;
}

export function readProgress(): ProgressState {
  if (typeof window === 'undefined') {
    return createDefaultProgress();
  }

  try {
    const stored = window.localStorage.getItem(STORAGE_KEY);
    return stored ? normalizeProgress(JSON.parse(stored)) : createDefaultProgress();
  } catch {
    return createDefaultProgress();
  }
}

export function saveProgress(next: ProgressState): void {
  if (typeof window === 'undefined') {
    return;
  }

  try {
    window.localStorage.setItem(STORAGE_KEY, JSON.stringify(normalizeProgress(next)));
  } catch {
    // Storage can be unavailable in private browsing or when its quota is exceeded.
  }
}

export function resetProgress(): ProgressState {
  const progress = createDefaultProgress();
  saveProgress(progress);
  return progress;
}

export function toggleLesson(
  state: ProgressState,
  id: LessonId,
  level: Level,
): ProgressState {
  const lessons = getLessonsForLevel(level);
  if (!lessons.some((lesson) => lesson.id === id)) {
    throw new Error(`Unknown lesson "${id}" for ${level} level.`);
  }

  const completedForLevel = state.completed[level];
  const isCompleted = completedForLevel.includes(id);
  const completed = isCompleted
    ? completedForLevel.filter((lessonId) => lessonId !== id)
    : [...completedForLevel, id];

  return {
    ...state,
    level,
    completed: {
      ...state.completed,
      [level]: completed,
    },
    lastLessonId: id,
  };
}

export function recordQuiz(
  state: ProgressState,
  level: Level,
  score: number,
): ProgressState {
  const integerScore = Number.isNaN(score) ? 0 : Math.trunc(score);
  const clampedScore = Math.max(0, Math.min(5, integerScore));

  return {
    ...state,
    level,
    quizScores: {
      ...state.quizScores,
      [level]: clampedScore,
    },
  };
}


===== src/hooks/useProgress.ts (new file) =====
import { useCallback, useEffect, useState } from 'react';
import {
  readProgress,
  recordQuiz as recordQuizProgress,
  resetProgress,
  saveProgress,
  toggleLesson as toggleLessonProgress,
} from '../lib/progress';
import type { LessonId, Level, ProgressState } from '../types/course';

export function useProgress() {
  const [progress, setProgress] = useState<ProgressState>(() => readProgress());

  useEffect(() => {
    saveProgress(progress);
  }, [progress]);

  const setLevel = useCallback((level: Level) => {
    setProgress((current) => ({
      ...current,
      level,
    }));
  }, []);

  const toggleLesson = useCallback((id: LessonId, level: Level) => {
    setProgress((current) => toggleLessonProgress(current, id, level));
  }, []);

  const recordQuiz = useCallback((level: Level, score: number) => {
    setProgress((current) => recordQuizProgress(current, level, score));
  }, []);

  const reset = useCallback(() => {
    setProgress(resetProgress());
  }, []);

  return {
    progress,
    level: progress.level,
    setLevel,
    toggleLesson,
    recordQuiz,
    reset,
  };
}

import type { Level } from '../types/course';

export interface LevelSwitcherProps {

  { value: 'certificate', label: 'ปวช.' },

    <div

          </button>
        );

    </div>
  );
}


===== src/components/ProgressBar.tsx (new file) =====
import type { LessonId, LessonSummary } from '../types/course';

export interface ProgressBarProps {
  lessons: LessonSummary[];
  completed: LessonId[];
}

export function ProgressBar({ lessons, completed }: ProgressBarProps) {
  const completedIds = new Set(completed);
  const completedCount = lessons.filter((lesson) => completedIds.has(lesson.id)).length;
  const percentage =
    lessons.length === 0 ? 0 : Math.round((completedCount / lessons.length) * 100);
  const valueText = `ทำเสร็จแล้ว ${completedCount} จาก ${lessons.length} บท (${percentage}%)`;

  return (
    <div className="w-full">
      <div className="mb-2 flex items-center justify-between gap-3 text-sm">
        <span className="font-semibold text-ink">ความคืบหน้า</span>
        <span className="text-muted">
          {completedCount}/{lessons.length} บท · {percentage}%
        </span>
      </div>
      <div
        role="progressbar"
        aria-label="ความคืบหน้า"
        aria-valuemin={0}
        aria-valuemax={100}
        aria-valuenow={percentage}
        aria-valuetext={valueText}
        className="h-2.5 overflow-hidden rounded-full bg-line"
      >
        <div
          className="h-full rounded-full bg-brand transition-[width] duration-300"
          style={{ width: `${percentage}%` }}
        />
      </div>
    </div>
  );
}

        certificate: ['overview'],
        diploma: ['database'],

    expect(readProgress()).toEqual(defaultProgress());
  });

  it('normalizes duplicate IDs, unknown IDs, and non-array completion data', () => {
    localStorage.setItem(
      STORAGE_KEY,
      JSON.stringify({
        version: 1,
        level: 'diploma',
        completed: {
          certificate: ['overview', 'overview', 'security', 'unknown'],
          diploma: 'database',
        },
        quizScores: {
          certificate: 4,
        },
        lastLessonId: 'api',
      }),
    );

    expect(readProgress()).toEqual({
      version: 1,
      level: 'diploma',
      completed: {
        certificate: ['overview'],
        diploma: [],
      },
      quizScores: {
        certificate: 4,
      },
      lastLessonId: 'api',
    });
  });

  it('drops invalid stored quiz scores and lesson references', () => {
    localStorage.setItem(
      STORAGE_KEY,
      JSON.stringify({
        ...defaultProgress(),
        quizScores: {
          certificate: 9,
          diploma: 2.5,
        },
        lastLessonId: 'security',
      }),
    );

    expect(readProgress()).toEqual(defaultProgress());
  });

  it('returns defaults and clears persisted progress on reset', () => {
    saveProgress({
      ...defaultProgress(),
      level: 'diploma',
      completed: {
        certificate: ['overview'],
        diploma: ['security'],
      },
      quizScores: {
        diploma: 5,
      },
    });

    const reset = resetProgress();

    expect(reset).toEqual(defaultProgress());
    expect(JSON.parse(localStorage.getItem(STORAGE_KEY) ?? 'null')).toEqual(defaultProgress());
    expect(readProgress()).toEqual(defaultProgress());
  });

  it('guards calls when browser storage is unavailable', () => {
    vi.stubGlobal('window', undefined);

    try {
      expect(readProgress()).toEqual(defaultProgress());
      expect(() => saveProgress(defaultProgress())).not.toThrow();
      expect(resetProgress()).toEqual(defaultProgress());
    } finally {
      vi.unstubAllGlobals();
    }
  });

  it('guards localStorage access errors', () => {
    const getItem = vi.spyOn(Storage.prototype, 'getItem').mockImplementation(() => {
      throw new Error('storage denied');
    });
    const setItem = vi.spyOn(Storage.prototype, 'setItem').mockImplementation(() => {
      throw new Error('storage denied');
    });

    try {
      expect(readProgress()).toEqual(defaultProgress());
      expect(() => saveProgress(defaultProgress())).not.toThrow();
      expect(() => resetProgress()).not.toThrow();
    } finally {
      getItem.mockRestore();
      setItem.mockRestore();
    }
  });
});

import { fireEvent, render, screen } from '@testing-library/react';

beforeEach(() => {
  localStorage.clear();
});

describe('LevelSwitcher', () => {
  it('renders the exact track labels with accessible selected state', () => {
    const onChange = vi.fn();
    const { rerender } = render(
      <LevelSwitcher level="certificate" onChange={onChange} />,
    );

    const certificate = screen.getByRole('button', { name: 'ปวช.' });
    const diploma = screen.getByRole('button', { name: 'ปวส.' });

    expect(certificate).toHaveAttribute('aria-pressed', 'true');
    expect(certificate).toHaveClass('bg-ink', 'text-surface');
    expect(diploma).toHaveAttribute('aria-pressed', 'false');

    fireEvent.click(diploma);

    expect(onChange).toHaveBeenCalledWith('diploma');

    rerender(<LevelSwitcher level="diploma" onChange={onChange} />);

    expect(certificate).toHaveAttribute('aria-pressed', 'false');
    expect(diploma).toHaveAttribute('aria-pressed', 'true');
    expect(diploma).toHaveClass('bg-ink', 'text-surface');
  });

  it('switches the persisted level without clearing either track progress', () => {
    const initial: ProgressState = {
      version: 1,
      level: 'certificate',
      completed: {
        certificate: ['overview', 'api'],
        diploma: ['security'],
      },
      quizScores: {
        certificate: 4,
        diploma: 5,
      },
      lastLessonId: 'api',
    };
    saveProgress(initial);

    function ProgressHarness() {
      const { level, setLevel } = useProgress();

      return <LevelSwitcher level={level} onChange={setLevel} />;
    }

    render(<ProgressHarness />);
    fireEvent.click(screen.getByRole('button', { name: 'ปวส.' }));

    expect(screen.getByRole('button', { name: 'ปวส.' })).toHaveAttribute(
      'aria-pressed',
      'true',
    );
    expect(readProgress()).toEqual({
      ...initial,
      level: 'diploma',
    });

import { render, screen } from '@testing-library/react';
import { describe, expect, it } from 'vitest';
import { getLessonsForLevel } from '../data/course';
import type { LessonId } from '../types/course';
import { ProgressBar } from './ProgressBar';

describe('ProgressBar', () => {
  it('exposes an accessible Thai progressbar with a rounded integer percentage', () => {
    const lessons = getLessonsForLevel('certificate').slice(0, 3);

    render(<ProgressBar lessons={lessons} completed={['overview']} />);

    const progressbar = screen.getByRole('progressbar', { name: 'ความคืบหน้า' });
    expect(progressbar).toHaveAttribute('aria-valuemin', '0');
    expect(progressbar).toHaveAttribute('aria-valuemax', '100');
    expect(progressbar).toHaveAttribute('aria-valuenow', '33');
    expect(progressbar).toHaveAttribute('aria-valuetext', 'ทำเสร็จแล้ว 1 จาก 3 บท (33%)');
  });

  it('counts each matching lesson once and ignores IDs outside the course', () => {
    const lessons = getLessonsForLevel('certificate');
    const completed: LessonId[] = [
      'overview',
      'overview',
      'security',
      'unknown' as LessonId,
    ];

    render(<ProgressBar lessons={lessons} completed={completed} />);

    expect(screen.getByRole('progressbar', { name: 'ความคืบหน้า' })).toHaveAttribute(
      'aria-valuenow',
      '13',
    );
  });

  it('reports zero rather than NaN for an empty lesson list', () => {
    render(<ProgressBar lessons={[]} completed={[]} />);

    expect(screen.getByRole('progressbar', { name: 'ความคืบหน้า' })).toHaveAttribute(
      'aria-valuenow',
      '0',
    );
  });
});
