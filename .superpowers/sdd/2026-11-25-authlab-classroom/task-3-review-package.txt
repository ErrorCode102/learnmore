import { useCallback, useEffect, useState } from 'react';
import {
  readProgress,
  recordQuiz as recordQuizProgress,
  resetProgress,
  saveProgress,
  toggleLesson as toggleLessonProgress,
} from '../lib/progress';
import type { LessonId, Level, ProgressState } from '../types/course';

export function useProgress() {
  const [progress, setProgress] = useState<ProgressState>(() => readProgress());

  useEffect(() => {
    saveProgress(progress);
  }, [progress]);

  const setLevel = useCallback((level: Level) => {
    setProgress((current) => ({
      ...current,
      level,
    }));
  }, []);

  const toggleLesson = useCallback((id: LessonId, level: Level) => {
    setProgress((current) => toggleLessonProgress(current, id, level));
  }, []);

  const visitLesson = useCallback((id: LessonId) => {
    setProgress((current) =>
      current.lastLessonId === id ? current : { ...current, lastLessonId: id },
    );
  }, []);

  const recordQuiz = useCallback((level: Level, score: number) => {
    setProgress((current) => recordQuizProgress(current, level, score));
  }, []);

  const reset = useCallback(() => {
    setProgress(resetProgress());
  }, []);

  return {
    progress,
    level: progress.level,
    setLevel,
    toggleLesson,
    visitLesson,
    recordQuiz,
    reset,
  };
}

import { createContext, useContext } from 'react';
import type { PropsWithChildren } from 'react';
import { useProgress } from '../hooks/useProgress';

type ClassroomProgress = ReturnType<typeof useProgress>;

const ProgressContext = createContext<ClassroomProgress | undefined>(undefined);

export function ProgressProvider({ children }: PropsWithChildren) {
  const progress = useProgress();

  return <ProgressContext.Provider value={progress}>{children}</ProgressContext.Provider>;
}

export function useClassroomProgress(): ClassroomProgress {
  const progress = useContext(ProgressContext);

  if (!progress) {
    throw new Error('useClassroomProgress must be used within a ProgressProvider.');
  }

  return progress;
}

import type { LessonId, LessonSummary } from '../types/course';

export interface ContinueSelection {
  lesson: LessonSummary;
  label: 'เริ่มบทแรก' | 'เรียนต่อ' | 'ทบทวนบทสุดท้าย';
}

export function getContinueSelection(
  lessons: LessonSummary[],
  completed: LessonId[],
): ContinueSelection | undefined {
  const finalLesson = lessons.at(-1);
  if (!finalLesson) {
    return undefined;
  }

  const completedIds = new Set(completed);
  const allComplete = lessons.every((lesson) => completedIds.has(lesson.id));
  const lesson =
    lessons.find((candidate) => !completedIds.has(candidate.id)) ?? finalLesson;

  if (allComplete) {
    return { lesson: finalLesson, label: 'ทบทวนบทสุดท้าย' };
  }

  return {
    lesson,
    label: completed.length === 0 ? 'เริ่มบทแรก' : 'เรียนต่อ',
  };
}

  lessons: LessonSummary[];
}

export function ContinueCard({ lessons }: ContinueCardProps) {

              >
                <ArrowLeft aria-hidden="true" className="h-4 w-4" />
                บทก่อนหน้า: {previousLesson.shortTitle}
              </Link>
            ) : (
              <Link
                to="/#course-map"
                className="inline-flex min-h-12 items-center justify-center gap-2 rounded-xl border border-line bg-surface px-5 py-3 text-sm font-semibold text-ink transition-colors hover:border-brand/30 hover:text-brand"
              >
                <ArrowLeft aria-hidden="true" className="h-4 w-4" />
                กลับไปยังแผนภูมิหลักสูตร
              </Link>
            )}

            {nextLesson ? (
              <Link
                to={`/lesson/${nextLesson.id}`}
                className="inline-flex min-h-12 items-center justify-center gap-2 rounded-xl bg-ink px-5 py-3 text-sm font-semibold text-white transition-colors hover:bg-brand-deep"
              >
                บทถัดไป: {nextLesson.shortTitle}
                <ArrowRight aria-hidden="true" className="h-4 w-4" />
              </Link>
            ) : (
              <Link
                to="/practice"
                className="inline-flex min-h-12 items-center justify-center gap-2 rounded-xl bg-ink px-5 py-3 text-sm font-semibold text-white transition-colors hover:bg-brand-deep"
              >
                ไปยังฝึกท้ายบท
                <ArrowRight aria-hidden="true" className="h-4 w-4" />
              </Link>
            )}
          </nav>
        </div>
      </div>
    </div>
  );
}
